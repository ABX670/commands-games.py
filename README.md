active_games = {}

def can_play(db, sender):
    from database import get_user
    s = db["settings"]
    u = get_user(db, sender)
    if u.get("banned_game"): return False
    if s.get("games_locked") and not u.get("is_admin"): return False
    return True

def handle_games(db, sender, text):
    t = text.strip()
    if not can_play(db, sender):
        return "⛔ أنت محظور من الألعاب أو الألعاب مقفولة للمشرفين فقط"

    if t == "تفكيك":
        name = random.choice(ANIME)
        active_games[sender] = ("تفكيك", name)
        return f"🧩 فكك الاسم: {name}\nاكتب الحروف مفككة: {' '.join(list(name))}"
    if t == "تركيب":
        name = random.choice(ANIME)
        active_games[sender] = ("تركيب", name)
        return f"🧩 ركب الاسم: {' '.join(list(name))}"
    if t == "كتابة":
        s = "السرعة سر النجاح"
        active_games[sender] = ("كتابة", s)
        return f"⌨️ اكتب الجملة بنفس الشكل بسرعة:\n{s}"
    if t == "اعلام":
        flag, country = random.choice(list(COUNTRIES.items()))
        active_games[sender] = ("اعلام", country)
        return f"{flag} ما اسم هذه الدولة؟"
    if t == "اسرع":
        e = random.choice(["😂","🔥","🍕","🚀","🐱","⚡","👑"])
        active_games[sender] = ("اسرع", e)
        return f"⚡ أسرع واحد يبعت هذا الايموجي: {e}"
    if t == "رياضيات":
        a = random.randint(1, 99); b = random.randint(1, 99)
        ans = str(a + b)
        active_games[sender] = ("رياضيات", ans)
        return f"🔢 حل بسرعة: {a} + {b} = ؟"
    if t == "خمن":
        n = random.randint(1, 20)
        active_games[sender] = ("خمن", str(n))
        return "🎯 خمنت رقم من 1 لـ 20، اكتب تخمينك"
    if any(x in t for x in ["حجرة","ورقة","مقص"]):
        choices = ["حجرة","ورقة","مقص"]
        bot = random.choice(choices)
        player = None
        for c in choices:
            if c in t: player = c; break
        if not player: return "اكتب: حجرة أو ورقة أو مقص"
        if player == bot: return f"🤝 تعادل! البوت اختار {bot}"
        win = (player=="حجرة" and bot=="مقص") or (player=="ورقة" and bot=="حجرة") or (player=="مقص" and bot=="ورقة")
        from database import get_user
        u = get_user(db, sender)
        if win:
            u["wins"] = u.get("wins",0)+1; u["xp"] = u.get("xp",0)+15; u["balance"] = u.get("balance",0)+15
            return f"🎉 فزت! البوت اختار {bot} +15$"
        else:
            return f"😢 خسرت! البوت اختار {bot}"

    if sender in active_games:
        typ, answer = active_games[sender]
        if text.replace(" ","").strip() == answer.replace(" ","").strip() or text.strip() == answer:
            del active_games[sender]
            from database import get_user
            u = get_user(db, sender)
            u["wins"] = u.get("wins",0)+1; u["xp"] = u.get("xp",0)+20; u["balance"] = u.get("balance",0)+10
            # XP boost check
            if u.get("xp_boost_until"):
                try:
                    from datetime import datetime
                    if datetime.now() < datetime.fromisoformat(u["xp_boost_until"]):
                        u["xp"] += 20
                except: pass
            return f"🎉 صح! فزت +10$ و +20 XP"
    return None
